# 1. compose 파일 위치 확정
ls -la ~/open-webui/
cat ~/open-webui/docker-compose.yml 2>/dev/null; echo "---"; cat ~/open-webui/compose.yml 2>/dev/null
ls -la ~/gitea-docker/
cat ~/gitea-docker/docker-compose.yml 2>/dev/null | head -n 150

# 2. 백업 - 0.11.0 -> 0.11.3 필수
docker exec open-webui tar czf /tmp/backup-0.11.0.tar.gz /app/backend/data 2>/dev/null
docker cp open-webui:/tmp/backup-0.11.0.tar.gz ~/open-webui-backup-$(date +%F).tar.gz
ls -lh ~/open-webui-backup-*.tar.gz

# 3. open-webui compose 수정
cd ~/open-webui
cat docker-compose.yml

# 이렇게 수정: image를 0.11.3으로 먼저
# image: ghcr.io/open-webui/open-webui:0.11.3
# volumes에 볼트 ro 마운트 추가 (MCP 서버가 호스트에서 볼트를 읽으니 open-webui도 같은 경로로 ro 마운트 필요)
# volumes:
#   - open-webui:/app/backend/data
#   - /mnt/build-data/oci-vaults/00_newbee:/mnt/build-data/oci-vaults/00_newbee:ro

# 1차 업그레이드 0.11.3
docker compose pull
docker compose down && docker compose up -d
docker logs -f open-webui --tail 100
# 8090 접속 확인 후

# 4. 0.12+ (main)으로 2차 업그레이드
# docker-compose.yml 에서
# image: ghcr.io/open-webui/open-webui:main  로 변경
docker compose pull
docker compose down && docker compose up -d
docker ps
docker exec open-webui cat /app/package.json | grep version
