python3 <<'EOF'
import json

p = ".obsidian/plugins/obsidian-git/data.json"
with open(p) as f:
    d = json.load(f)

# 자동화 설정
d["autoSaveInterval"] = 30          # 30초마다 로컬 커밋
d["autoPushInterval"] = 60          # 1분마다 Gitea로 push
d["autoPullInterval"] = 60          # 1분마다 Gitea에서 pull
d["autoPullOnBoot"] = True          # Obsidian 시작 시 pull
d["autoBackupAfterFileChange"] = True
d["disablePush"] = False
d["pullBeforePush"] = True

with open(p, "w") as f:
    json.dump(d, f, indent=2, ensure_ascii=False)

print("업데이트 완료")
for k in ["autoSaveInterval","autoPushInterval","autoPullInterval","autoPullOnBoot"]:
    print(f"  {k}: {d[k]}")
EOF
