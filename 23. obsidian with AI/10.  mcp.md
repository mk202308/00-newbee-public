```
# 1. 디렉토리 생성 및 권한
sudo mkdir -p /opt/mcp-obsidian
sudo chown ubuntu:ubuntu /opt/mcp-obsidian
cd /opt/mcp-obsidian

# 2. package.json - v1 라인 최종 고정 (zod deduped 중요)
cat > package.json <<'JSON'
{
  "type": "module",
  "dependencies": {
    "@modelcontextprotocol/sdk": "1.17.3",
    "express": "4.21.2",
    "cors": "2.8.5",
    "zod": "3.25.76"
  }
}
JSON

# 3. server.js - 2026-07-28 stateless + McpServer API (setRequestHandler 아님)
cat > server.js <<'JS'
#!/usr/bin/env node
// MCP SDK 1.17.3 pinned - v1 line, 2026-07-28 stateless - McpServer API
import { McpServer } from "@modelcontextprotocol/sdk/server/mcp.js";
import { StreamableHTTPServerTransport } from "@modelcontextprotocol/sdk/server/streamableHttp.js";
import express from "express";
import cors from "cors";
import fs from "fs/promises";
import path from "path";
import { z } from "zod";

const VAULT = "/mnt/build-data/oci-vaults/00_newbee";
const PORT = 8001;
const SDK_VERSION = "1.17.3";
const PROTOCOL_VERSION = "2026-07-28";

const app = express();
app.use(cors({ exposedHeaders: ["mcp-session-id", "mcp-protocol-version"] }));
app.use(express.json({ limit: "5mb" }));

const mcpServer = new McpServer(
  { name: "obsidian-00_newbee", version: "1.0.0" },
  { capabilities: { tools: {} } }
);

const safe = (p) => path.join(VAULT, (p||"").replace(/^\//, ""));

mcpServer.registerTool("search", {
  description: "vault md 검색",
  inputSchema: { query: z.string().describe("검색어") }
}, async ({ query }) => {
  const q = (query||"").toLowerCase();
  const results = [];
  async function walk(d) {
    for (const e of await fs.readdir(d, { withFileTypes: true })) {
      if (e.name.startsWith(".") || e.name === ".git" || e.name === ".obsidian") continue;
      const full = path.join(d, e.name);
      if (e.isDirectory()) await walk(full);
      else if (e.name.endsWith(".md")) {
        const txt = await fs.readFile(full, "utf-8").catch(() => "");
        if (txt.toLowerCase().includes(q)) results.push(path.relative(VAULT, full));
        if (results.length > 50) return;
      }
    }
  }
  await walk(VAULT);
  return { content: [{ type: "text", text: results.join("\n") || "결과 없음" }] };
});

mcpServer.registerTool("read", {
  description: "파일 읽기",
  inputSchema: { path: z.string() }
}, async ({ path: p }) => {
  const text = await fs.readFile(safe(p), "utf-8");
  return { content: [{ type: "text", text: text.slice(0, 20000) }] };
});

mcpServer.registerTool("list", {
  description: "폴더 목록",
  inputSchema: { dir: z.string().optional() }
}, async ({ dir }) => {
  const d = safe(dir || "");
  const files = await fs.readdir(d, { withFileTypes: true });
  return { content: [{ type: "text", text: files.map(f => (f.isDirectory()? "📁 " : "📄 ") + f.name).join("\n") }] };
});

mcpServer.registerTool("write", {
  description: "파일 쓰기",
  inputSchema: { path: z.string(), content: z.string() }
}, async ({ path: p, content }) => {
  const full = safe(p);
  await fs.mkdir(path.dirname(full), { recursive: true });
  await fs.writeFile(full, content, "utf-8");
  return { content: [{ type: "text", text: `저장 완료: ${p}` }] };
});

app.post("/mcp", async (req, res) => {
  const transport = new StreamableHTTPServerTransport({ sessionIdGenerator: undefined });
  await mcpServer.connect(transport);
  await transport.handleRequest(req, res, req.body);
});

app.get("/health", (req,res) => res.json({
  ok: true,
  vault: VAULT,
  endpoint: "/mcp",
  sdk: SDK_VERSION,
  protocol: PROTOCOL_VERSION,
  mode: "mcpserver-stateless-2026-07-28",
  tools: 4
}));

app.listen(PORT, "0.0.0.0", () => console.log(`MCP ${SDK_VERSION} McpServer ${PROTOCOL_VERSION} stateless on ${PORT}/mcp`));
JS

# 4. 설치 - zod deduped 확인 필수
rm -rf node_modules package-lock.json
npm install --save-exact
npm list zod @modelcontextprotocol/sdk
# 결과: zod@3.25.76 deduped 단일 버전이어야 정상

# 5. systemd 서비스 등록
NODE_BIN=$(which node)
echo "node: $NODE_BIN"

sudo tee /etc/systemd/system/mcp-obsidian.service > /dev/null <<EOF
[Unit]
Description=MCP Obsidian v1 1.17.3 2026-07-28 stateless - 00_newbee
After=network.target

[Service]
User=ubuntu
WorkingDirectory=/opt/mcp-obsidian
ExecStart=${NODE_BIN} /opt/mcp-obsidian/server.js
Restart=always
Environment=PORT=8001
ReadWritePaths=/mnt/build-data/oci-vaults/00_newbee

[Install]
WantedBy=multi-user.target
EOF

sudo systemctl daemon-reload
sudo systemctl enable --now mcp-obsidian.service

# 6. 확인
sleep 3
sudo systemctl status mcp-obsidian.service --no-pager -l
sudo journalctl -u mcp-obsidian.service -n 20 --no-pager
curl -s http://localhost:8001/health | python3 -m json.tool
```


이제 `1.17.3` + `zod@3.25.76 deduped` + `McpServer API` 조합으로 고정 성공이다. 앞에 크래시는 `Server.setRequestHandler` 구문법 때문이었고 지금 `McpServer.registerTool`로 가니까 정상 기동.

앞에 `curl`이 `Expecting value: line 1 column 1` 뜬 것도 서버가 죽어있어서 HTML 에러 페이지만 받아서 JSON 파싱 실패한 거다. 지금은 살아있다.
